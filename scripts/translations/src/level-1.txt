def two_sum(numbers, target):
    number_to_index = {}
    for number_index in range(len(numbers)):
        need = target - numbers[number_index]
        if need in number_to_index:
            return [number_to_index[need], number_index]
.
        number_to_index[numbers[number_index]] = number_index
.
    return []

@java
static int[] twoSum(int[] numbers, int target) {
    Map<Integer, Integer> numberToIndex = new HashMap<>();
    for (int numberIndex = 0; numberIndex < numbers.length; numberIndex++) {
        int need = target - numbers[numberIndex];
        if (numberToIndex.containsKey(need)) {
            return new int[]{numberToIndex.get(need), numberIndex};
        }
        numberToIndex.put(numbers[numberIndex], numberIndex);
    }
    return new int[]{};
}

        return False

@java
static boolean isAnagram(String firstWord, String secondWord) {
    if (firstWord.length() != secondWord.length()) {
        return false;
    }
    Map<Character, Integer> count = new HashMap<>();
    for (int firstWordIndex = 0; firstWordIndex < firstWord.length(); firstWordIndex++) {
        count.put(firstWord.charAt(firstWordIndex), count.getOrDefault(firstWord.charAt(firstWordIndex), 0) + 1);
    }
    for (int secondWordIndex = 0; secondWordIndex < secondWord.length(); secondWordIndex++) {
        int remaining = count.getOrDefault(secondWord.charAt(secondWordIndex), 0);
        if (remaining == 0) {
            return false;
        }
        count.put(secondWord.charAt(secondWordIndex), remaining - 1);
    }
    return true;
}
