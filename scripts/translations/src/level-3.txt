static List<List<Integer>> levelOrder(TreeNode treeRoot) {
    List<List<Integer>> result = new ArrayList<>();
    List<TreeNode> queue = new ArrayList<>(List.of(treeRoot));
    int queueFrontIndex = 0;
    while (queueFrontIndex < queue.size()) {
        int size = queue.size() - queueFrontIndex;
        List<Integer> level = new ArrayList<>();
        for (int levelPosition = 0; levelPosition < size; levelPosition++) {
            TreeNode currentNode = queue.get(queueFrontIndex++);
            level.add(currentNode.val);
            if (currentNode.left != null) queue.add(currentNode.left);
            if (currentNode.right != null) queue.add(currentNode.right);
        }
        result.add(level);
    }
    return result;
}
